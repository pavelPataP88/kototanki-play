function updateParticles(dt){
  state.hearts = state.hearts.filter(p=>{p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy-=20*dt; return p.life>0;});
  state.spark = state.spark.filter(p=>{p.life-=dt; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.97; p.vy*=0.97; return p.life>0;});
}

function update(dt){
  const s=scenes[scene];
  updateParticles(dt);
  if(state.kissTimer>0){
    state.kissTimer-=dt; player.vx=0; player.anim=0; const center=(player.x+nat.x)/2; player.x += (center-42-player.x)*Math.min(1,dt*6); nat.x += (center+42-nat.x)*Math.min(1,dt*6); player.dir=1; nat.dir=-1;

      const handGap=state.handHold?68:108;
      const target=player.x+(player.dir>0?handGap:-handGap); nat.x += (target-nat.x)*Math.min(1,dt*(state.handHold?6:4.5)); nat.dir=player.dir<0?1:-1;

  }
  if(s.kind==='walk'||s.kind==='final'){state.handHold=true; nat.follow=true; nat.visible=true;}
  state.energy += (Math.abs(player.vx)>0? -3.5: 5)*dt; state.energy=clamp(state.energy,25,100); updateHud();

  // special cutscene positions
  if(state.inCar){
    drawCar(carX,600);
    drawSprite('m4',carX-45,548,190,false,1,Math.sin(t*8)*1.5);
    drawSprite('f4',carX+40,555,165,true,1,Math.sin(t*8+1)*1.5);
  } else {
    let pKey='m0', fKey='f0';
    const moving=Math.abs(player.vx)>2;
    if(state.stage) pKey='m5';
    else if(state.danceTimer>0) {pKey='m5'; fKey='f5';}
    else if(state.kissTimer>0) {pKey='m3'; fKey='f3';}
    else pKey=moving?('m'+(1+(Math.floor(player.anim)%2))):'m0';
    if(!state.kissTimer && !state.danceTimer && nat.follow) fKey=state.handHold?('f'+(1+(Math.floor(player.anim+.7)%2))):'f0';

    // Natasha static before interactions
    if(s.kind==='city' && !state.met){ drawSprite('f0',s.natasha,595,260,false,1,Math.sin(t*2)*2); }
    else if((s.kind==='flowers'||s.kind==='dance') && !state.handHold){ drawSprite(state.danceTimer>0?'f5':'f0',s.natasha,595,260,false,1,Math.sin(t*2)*2); }

    drawSprite(pKey,player.x,605,(state.stage?310:(state.danceTimer>0?315:300)),player.dir<0,1,Math.sin(t*6+1)*1.5);

    if(s.kind==='army'){ drawRifle(player.x+(player.dir<0?-40:35),530,player.dir<0); }
    if(s.kind==='walk'){ drawRifle(player.x+(player.dir<0?-40:35),530,player.dir<0); }
    if(state.flowers && (s.kind==='flowers' || s.kind==='dance')) drawBouquet(player.x+(player.dir<0?-22:22),545,.85);
    if(state.handHold && nat.visible && !state.kissTimer && !state.danceTimer && !state.inCar){ drawLinkedHands(player.x+(player.dir<0?-36:36),510,nat.x+(player.dir<0?34:-34),510); }
  }

  // scene overlays
  if(s.kind==='city' && state.met && !state.kissed && !state.inCar){ ctx.fillStyle='rgba(255,120,170,.1)'; ctx.fillRect(0,0,1280,720); }
  if(state.kissTimer>0){ ctx.fillStyle='rgba(255,120,170,.08)'; ctx.fillRect(0,0,1280,720); }
  if(state.danceTimer>0){ ctx.fillStyle=`rgba(255,210,110,${0.06+0.04*Math.sin(t*6)})`; ctx.fillRect(0,0,1280,720); }
  if(state.stage){ ctx.fillStyle=`rgba(255,200,80,${.09+.08*Math.sin(t*6)})`; ctx.fillRect(0,0,1280,720); }
  drawParticles();
  ctx.restore();
}
